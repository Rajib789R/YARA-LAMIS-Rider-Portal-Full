from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.worksheet.datavalidation import DataValidation
from openpyxl.chart import BarChart, Reference
from openpyxl.formatting.rule import CellIsRule
from openpyxl.utils import get_column_letter

FILE_NAME = "YARA_LAMIS_Rider_Smart_Ledger_35.xlsx"

wb = Workbook()

# =========================================================
# STYLES
# =========================================================
blue = "1F4E78"
light_blue = "D9EAF7"
green = "70AD47"
red = "C00000"
orange = "F4B183"
white = "FFFFFF"
gray = "E7E6E6"

header_fill = PatternFill("solid", fgColor=blue)
sub_fill = PatternFill("solid", fgColor=light_blue)
green_fill = PatternFill("solid", fgColor=green)
red_fill = PatternFill("solid", fgColor=red)
orange_fill = PatternFill("solid", fgColor=orange)

header_font = Font(color=white, bold=True)
title_font = Font(size=18, bold=True)
bold = Font(bold=True)

thin = Side(style="thin", color="B7B7B7")
border = Border(left=thin, right=thin, top=thin, bottom=thin)

# =========================================================
# 1. DASHBOARD
# =========================================================
dash = wb.active
dash.title = "Dashboard"

dash["A1"] = "YARA LAMIS"
dash["B1"] = "RIDER SMART LEDGER"
dash["A1"].font = Font(size=20, bold=True)
dash["B1"].font = Font(size=20, bold=True)

dashboard_data = [
    ("Total Riders", "=COUNTA('Rider Master'!A2:A36)"),
    ("Total Payments", '=SUMIF(Transactions!D2:D501,"Payment",Transactions!E2:E501)'),
    ("Total Expenses", '=SUMIF(Transactions!D2:D501,"Expense",Transactions!E2:E501)'),
    ("Total Outstanding", "=SUM('Rider Master'!H2:H36)"),
    ("Total Advance", '=SUMIF(Transactions!F2:F501,"Advance",Transactions!E2:E501)'),
    ("Total Penalty", '=SUMIF(Transactions!F2:F501,"Penalty",Transactions!E2:E501)'),
    ("Total Accident", '=SUMIF(Transactions!F2:F501,"Accident Payment",Transactions!E2:E501)'),
    ("Total Fuel", '=SUMIF(Transactions!F2:F501,"Fuel Extra",Transactions!E2:E501)'),
    ("Total Food", '=SUMIF(Transactions!F2:F501,"Monthly Food",Transactions!E2:E501)'),
    ("Total Mobile", '=SUMIF(Transactions!F2:F501,"Mobile Buy",Transactions!E2:E501)'),
]

dash["A3"] = "Dashboard Summary"
dash["A3"].fill = header_fill
dash["A3"].font = header_font

dash["A4"] = "Metric"
dash["B4"] = "Amount / Count"

for cell in dash[4]:
    cell.fill = header_fill
    cell.font = header_font
    cell.border = border

for row_num, (label, formula) in enumerate(dashboard_data, start=5):
    dash.cell(row_num, 1, label)
    dash.cell(row_num, 2, formula)

    dash.cell(row_num, 1).border = border
    dash.cell(row_num, 2).border = border

dash.column_dimensions["A"].width = 30
dash.column_dimensions["B"].width = 25

# =========================================================
# 2. RIDER MASTER
# =========================================================
rm = wb.create_sheet("Rider Master")

headers = [
    "Rider ID",
    "Name",
    "Iqama Number",
    "Mobile",
    "Supervisor",
    "Status",
    "Net Balance",
    "Outstanding"
]

for col, header in enumerate(headers, 1):
    cell = rm.cell(1, col, header)
    cell.fill = header_fill
    cell.font = header_font
    cell.border = border
    cell.alignment = Alignment(horizontal="center")

# 35 riders
for r in range(2, 37):

    # Net Balance = Expenses - Payments
    rm.cell(
        r,
        7,
        f'=SUMIFS(Transactions!$E:$E,Transactions!$B:$B,A{r},Transactions!$D:$D,"Expense")'
        f'-SUMIFS(Transactions!$E:$E,Transactions!$B:$B,A{r},Transactions!$D:$D,"Payment")'
    )

    rm.cell(r, 8, f"=G{r}")

    for c in range(1, 9):
        rm.cell(r, c).border = border

widths = [16, 25, 22, 18, 22, 15, 18, 18]

for i, width in enumerate(widths, 1):
    rm.column_dimensions[get_column_letter(i)].width = width

# Status dropdown
status_dropdown = DataValidation(
    type="list",
    formula1='"Active,Inactive,Suspended,Transferred"',
    allow_blank=True
)

rm.add_data_validation(status_dropdown)
status_dropdown.add("F2:F36")

# Outstanding red highlight
rm.conditional_formatting.add(
    "H2:H36",
    CellIsRule(
        operator="greaterThan",
        formula=["0"],
        fill=red_fill
    )
)

# =========================================================
# 3. TRANSACTIONS
# =========================================================
tr = wb.create_sheet("Transactions")

transaction_headers = [
    "Date",
    "Rider ID",
    "Name",
    "Entry Type",
    "Amount (SAR)",
    "Category",
    "Note",
    "Supervisor"
]

for col, header in enumerate(transaction_headers, 1):
    cell = tr.cell(1, col, header)
    cell.fill = header_fill
    cell.font = header_font
    cell.border = border
    cell.alignment = Alignment(horizontal="center")

# 500 transaction rows
for r in range(2, 502):

    # Automatic Rider Name
    tr.cell(
        r,
        3,
        f'=IFERROR(VLOOKUP(B{r},\'Rider Master\'!A:B,2,FALSE),"")'
    )

    # Automatic Supervisor
    tr.cell(
        r,
        8,
        f'=IFERROR(VLOOKUP(B{r},\'Rider Master\'!A:E,5,FALSE),"")'
    )

    for c in range(1, 9):
        tr.cell(r, c).border = border

# Dropdown: Payment / Expense
entry_dropdown = DataValidation(
    type="list",
    formula1='"Payment,Expense"',
    allow_blank=True
)

tr.add_data_validation(entry_dropdown)
entry_dropdown.add("D2:D501")

# Category dropdown
category_dropdown = DataValidation(
    type="list",
    formula1='"Advance,Mobile Buy,Fuel Extra,Monthly Food,Penalty,Traffic Violation,Accident Payment,Rider Transfer Amount,Maintenance,Other"',
    allow_blank=True
)

tr.add_data_validation(category_dropdown)
category_dropdown.add("F2:F501")

transaction_widths = [
    14, 16, 25, 16, 18, 25, 40, 22
]

for i, width in enumerate(transaction_widths, 1):
    tr.column_dimensions[get_column_letter(i)].width = width

tr.freeze_panes = "A2"
tr.auto_filter.ref = "A1:H501"

# =========================================================
# 4. RIDER PROFILE
# =========================================================
profile = wb.create_sheet("Rider Profile")

profile.column_dimensions["A"].width = 25

# =========================================================
# 5. MONTHLY REPORT
# =========================================================
monthly = wb.create_sheet("Monthly Report")

monthly_headers = [
    "Month",
    "Payments",
    "Expenses",
    "Outstanding"
]

for col, header in enumerate(monthly_headers, 1):
    cell = monthly.cell(1, col, header)
    cell.fill = header_fill
    cell.font = header_font
    cell.border = border

months = [
    "January", "February", "March", "April",
    "May", "June", "July", "August",
    "September", "October", "November", "December"
]

for i, month in enumerate(months, start=2):

    monthly.cell(i, 1, month)

    month_number = i - 1

    monthly.cell(
        i,
        2,
        f'=SUMPRODUCT((MONTH(Transactions!$A$2:$A$501)={month_number})'
        f'*(Transactions!$D$2:$D$501="Payment")'
        f'*Transactions!$E$2:$E$501)'
    )

    monthly.cell(
        i,
        3,
        f'=SUMPRODUCT((MONTH(Transactions!$A$2:$A$501)={month_number})'
        f'*(Transactions!$D$2:$D$501="Expense")'
        f'*Transactions!$E$2:$E$501)'
    )

    monthly.cell(i, 4, f"=C{i}-B{i}")

    for c in range(1, 5):
        monthly.cell(i, c).border = border

for i in range(1, 5):
    monthly.column_dimensions[get_column_letter(i)].width = 22

    "Supervisor Report",

    for row in sheet.iter_rows():
